**배선 (UNO ↔ MAX485)**

| MAX485 핀 | UNO 핀 |
|---|---|
| VCC | 5V |
| GND | GND |
| RO | D10 |
| DI | D11 |
| DE + RE (둘을 묶음) | D4 |
| A | 버스 A |
| B | 버스 B |

**PC ↔ 버스**

1. PC에 USB-RS485 컨버터를 꽂습니다(CH340/FT232 계열, 드라이버 설치).
2. 컨버터 A ↔ MAX485 A, 컨버터 B ↔ MAX485 B를 트위스트 페어로 연결합니다. 컨버터 표기가 D+/D−면 D+=A, D−=B입니다. A-B가 바뀌면 통신이 안 됩니다.
3. 컨버터 GND와 UNO GND를 연결합니다(기준 전위 공유).
4. 버스 양 끝(컨버터, MAX485)의 A-B 사이에 120Ω 종단저항을 각각 답니다. 1:1 짧은 거리면 생략해도 대개 동작하지만, 교육용 표준 배선으로는 넣는 걸 원칙으로 합니다.
5. UNO USB는 PC에 따로 연결합니다(전원 + 디버그 시리얼 115200).

**PC에서 확인**

- 장치 관리자에서 컨버터 COM 포트 번호 확인.
- QModMaster(GUI) 또는 pymodbus로 읽기: 슬레이브 ID 1, 9600 8N1, FC 03, 시작 주소 202(프로토콜 주소, =KS 203), 수량 2 → 2워드를 하위 워드 먼저 조합해 float으로 변환하면 온도.
- 먼저 주소 0부터 8워드(KS 1~8)를 읽어 3번째 값이 1(센서 노드), 6번째가 30이 나오면 맵과 배선이 맞는 겁니다.

pymodbus 예시 (3.x):

```python
from pymodbus.client import ModbusSerialClient
import struct

c = ModbusSerialClient(port="COM5", baudrate=9600, parity="N", stopbits=1, timeout=1)
c.connect()
r = c.read_holding_registers(address=202, count=2, slave=1)
lo, hi = r.registers
temp = struct.unpack("<f", struct.pack("<HH", lo, hi))[0]
print(f"temp = {temp:.2f} ℃")
c.close()
```

**안 될 때**

- 응답 없음: A/B 반전, GND 미연결, DE/RE 핀 오배선, COM 포트 오선택
- CRC 오류/깨짐: 통신속도 불일치, 종단저항 없이 긴 선
- 값이 이상함: 워드 순서(하위 먼저) 미적용 — QModMaster에선 "Little-endian word order" 옵션 선택







/*
 * ksx3267_sensor_node_uno.ino
 * KS X 3267 부속서 A.1 디폴트 레지스터 맵 — 센서 노드 (Modbus RTU 슬레이브)
 *
 * 하드웨어 : Arduino UNO + MAX485
 *            MAX485 RO -> D10 (SoftwareSerial RX)
 *            MAX485 DI -> D11 (SoftwareSerial TX)
 *            MAX485 DE+RE -> D4
 *            USB 시리얼(D0/D1)은 디버그 출력 전용
 * 통신     : 9600 bps, 8N1, 슬레이브 주소 1, FC 03/04 지원
 * 라이브러리: 없음 (SoftwareSerial 기본 포함, Modbus 슬레이브 직접 구현)
 *
 * 온도 1채널·습도 1채널 값은 실제 센서 대신 난수로 생성한다(교육·시험용).
 *
 * 주소 규칙 : KS 주소는 1-based, 프로토콜(전문) 주소 = KS 주소 - 1
 * float 인코딩(4.3.3): 워드 내 빅 엔디언, 워드 간 리틀 엔디언(하위 워드 먼저)
 */

#include <SoftwareSerial.h>

#define RS485_RX     10
#define RS485_TX     11
#define DE_RE_PIN    4
#define SLAVE_ID     1
#define BAUD         9600

SoftwareSerial rs485(RS485_RX, RS485_TX);

// ---- 레지스터 저장소: KS 주소 1~292 (인덱스 = 프로토콜 주소 = KS-1) ----
const uint16_t REG_COUNT = 292;
uint16_t regs[REG_COUNT];

inline void setReg(uint16_t ksAddr, uint16_t v) { regs[ksAddr - 1] = v; }

void setFloat(uint16_t ksAddr, float v) {      // 4.3.3 규칙
  uint32_t raw; memcpy(&raw, &v, 4);
  setReg(ksAddr,     raw & 0xFFFF);            // 하위 워드 먼저
  setReg(ksAddr + 1, raw >> 16);               // 상위 워드
}
void setU32(uint16_t ksAddr, uint32_t v) {
  setReg(ksAddr,     v & 0xFFFF);
  setReg(ksAddr + 1, v >> 16);
}

// ---- KS X 3267 A.1 노드 정보 상수 ----
const uint16_t ORG_CODE     = 0;   // 1 기관코드 (디폴트 맵 = 0)
const uint16_t COMPANY_CODE = 0;   // 2 회사코드 (디폴트 맵 = 0)
const uint16_t PRODUCT_TYPE = 1;   // 3 제품타입 (1 = 센서 노드)
const uint16_t PRODUCT_CODE = 0;   // 4 제품코드
const uint16_t PROTO_VER    = 10;  // 5 프로토콜 버전
const uint16_t CHANNELS     = 30;  // 6 채널수 (디폴트 맵 고정)
const uint32_t SERIAL_NO    = 1;   // 7~8 시리얼번호
const uint16_t DEV_TEMP     = 1;   // 101 온도 1 디바이스 코드
const uint16_t DEV_HUMI     = 2;   // 104 습도 1 디바이스 코드
const uint16_t STAT_READY   = 0;   // 202/205/214 상태 코드

// 난수 범위
const float TEMP_MIN = 18.0, TEMP_MAX = 32.0;   // ℃
const float HUMI_MIN = 40.0, HUMI_MAX = 90.0;   // %
const unsigned long UPDATE_MS = 1000;
unsigned long lastUpdate = 0;

// ---- Modbus RTU 프레임 처리 ----
uint8_t  frame[64];
uint8_t  frameLen = 0;
unsigned long lastByteAt = 0;
const unsigned long FRAME_GAP_MS = 4;           // 9600bps 3.5문자 ≈ 4ms

uint16_t crc16(const uint8_t *buf, uint8_t len) {
  uint16_t crc = 0xFFFF;
  for (uint8_t i = 0; i < len; i++) {
    crc ^= buf[i];
    for (uint8_t b = 0; b < 8; b++)
      crc = (crc & 1) ? (crc >> 1) ^ 0xA001 : crc >> 1;
  }
  return crc;
}

void sendFrame(uint8_t *buf, uint8_t len) {
  uint16_t crc = crc16(buf, len);
  buf[len++] = crc & 0xFF;                       // CRC는 하위 바이트 먼저
  buf[len++] = crc >> 8;
  digitalWrite(DE_RE_PIN, HIGH);                 // 송신 모드
  rs485.write(buf, len);
  rs485.flush();                                 // 전송 완료 대기 (중요)
  digitalWrite(DE_RE_PIN, LOW);                  // 수신 모드 복귀
}

void sendException(uint8_t fc, uint8_t code) {
  uint8_t r[5] = { SLAVE_ID, (uint8_t)(fc | 0x80), code };
  sendFrame(r, 3);
}

void handleFrame() {
  if (frameLen < 8 || frame[0] != SLAVE_ID) return;
  uint16_t crc = frame[frameLen - 2] | (frame[frameLen - 1] << 8);
  if (crc != crc16(frame, frameLen - 2)) return;

  uint8_t fc = frame[1];
  if (fc != 0x03 && fc != 0x04) { sendException(fc, 0x01); return; }

  uint16_t start = (frame[2] << 8) | frame[3];   // 프로토콜 주소 (0-based)
  uint16_t qty   = (frame[4] << 8) | frame[5];
  if (qty < 1 || qty > 125)                     { sendException(fc, 0x03); return; }
  if ((uint32_t)start + qty > REG_COUNT)        { sendException(fc, 0x02); return; }

  static uint8_t r[3 + 250 + 2];
  r[0] = SLAVE_ID; r[1] = fc; r[2] = qty * 2;
  for (uint16_t i = 0; i < qty; i++) {
    r[3 + i * 2]     = regs[start + i] >> 8;
    r[3 + i * 2 + 1] = regs[start + i] & 0xFF;
  }
  sendFrame(r, 3 + qty * 2);
}

void pollModbus() {
  while (rs485.available()) {
    if (frameLen < sizeof(frame)) frame[frameLen++] = rs485.read();
    else rs485.read();
    lastByteAt = millis();
  }
  if (frameLen > 0 && millis() - lastByteAt >= FRAME_GAP_MS) {
    handleFrame();
    frameLen = 0;
  }
}

float randomFloat(float lo, float hi) {
  return lo + (random(0, 10001) / 10000.0f) * (hi - lo);
}

void setup() {
  pinMode(DE_RE_PIN, OUTPUT);
  digitalWrite(DE_RE_PIN, LOW);
  Serial.begin(115200);
  rs485.begin(BAUD);
  randomSeed(analogRead(A0));

  memset(regs, 0, sizeof(regs));
  setReg(1, ORG_CODE);  setReg(2, COMPANY_CODE);
  setReg(3, PRODUCT_TYPE); setReg(4, PRODUCT_CODE);
  setReg(5, PROTO_VER); setReg(6, CHANNELS);
  setU32(7, SERIAL_NO);
  setReg(101, DEV_TEMP);                         // 미장착 채널(102~130)은 0
  setReg(104, DEV_HUMI);
  setReg(202, STAT_READY); setReg(205, STAT_READY); setReg(214, STAT_READY);

  Serial.println(F("KS X 3267 sensor node (UNO, random temp) ready, ID 1"));
}

void loop() {
  pollModbus();                                  // delay() 금지 — 스캔 정지

  if (millis() - lastUpdate >= UPDATE_MS) {
    lastUpdate = millis();
    float temp = randomFloat(TEMP_MIN, TEMP_MAX);
    float humi = randomFloat(HUMI_MIN, HUMI_MAX);
    setFloat(203, temp); setReg(205, STAT_READY); // 온도1 값·상태
    setFloat(212, humi); setReg(214, STAT_READY); // 습도1 값·상태
    Serial.print(F("temp=")); Serial.print(temp, 2);
    Serial.print(F("  humi=")); Serial.println(humi, 2);
  }
}
